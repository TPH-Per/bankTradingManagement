-- ============================================================================
-- Cassandra Schema for Bank Trading Management System
-- ============================================================================
-- Version: 2.0
-- Description: Complete schema for accounts, transactions, P2P transfers,
--              customer management, and audit logging
-- ============================================================================

-- ============================================================================
-- KEYSPACES (DEV Environment)
-- ============================================================================

CREATE KEYSPACE IF NOT EXISTS bank_trading
  WITH REPLICATION = {'class':'SimpleStrategy','replication_factor':1};

CREATE KEYSPACE IF NOT EXISTS bank_trading_audit
  WITH REPLICATION = {'class':'SimpleStrategy','replication_factor':1};

-- ============================================================================
-- MAIN KEYSPACE: bank_trading (Business Logic)
-- ============================================================================
USE bank_trading;

-- ----------------------------------------------------------------------------
-- 1. ACCOUNTS - Account Management
-- ----------------------------------------------------------------------------

-- Primary account table - lookup by account_id
CREATE TABLE IF NOT EXISTS accounts_by_id (
  account_id   text PRIMARY KEY,
  customer_id  text,
  currency     text,
  status       text,          -- ACTIVE/LOCKED/CLOSED
  opened_at    timestamp,
  updated_at   timestamp,
  extra_json   text
);

-- Account lookup by customer - supports multiple accounts per customer
CREATE TABLE IF NOT EXISTS accounts_by_customer (
  customer_id  text,
  account_id   text,
  opened_at    timestamp,
  status       text,
  PRIMARY KEY ((customer_id), account_id)
);

-- Account balances - real-time balance tracking
CREATE TABLE IF NOT EXISTS balances (
  account_id   text PRIMARY KEY,
  balance      decimal,
  reserved     decimal,        -- For transaction locking
  updated_at   timestamp
);

-- Daily balance snapshots - for statements and historical queries
CREATE TABLE IF NOT EXISTS balance_daily_snapshots (
  account_id      text,
  day             date,
  balance_open    decimal,
  balance_close   decimal,
  total_debit     decimal,
  total_credit    decimal,
  num_tx          int,
  updated_at      timestamp,
  PRIMARY KEY ((account_id), day)
) WITH CLUSTERING ORDER BY (day DESC);

-- ----------------------------------------------------------------------------
-- 2. CUSTOMERS - Customer Identity Management
-- ----------------------------------------------------------------------------

-- Lookup by national ID (CCCD/CMND/Passport)
CREATE TABLE IF NOT EXISTS customers_by_identity (
  national_id text PRIMARY KEY,
  customer_id text,
  full_name   text,
  dob         date,
  phone       text,
  status      text,               -- ACTIVE / LOCKED / CLOSED
  created_at  timestamp,
  updated_at  timestamp,
  extra_json  text
);

-- Lookup by email
CREATE TABLE IF NOT EXISTS customers_by_email (
  email       text PRIMARY KEY,
  customer_id text,
  full_name   text,
  phone       text,
  national_id text,
  status      text,
  created_at  timestamp,
  updated_at  timestamp,
  extra_json  text
);

-- Lookup by phone number
CREATE TABLE IF NOT EXISTS customers_by_phone (
  phone       text PRIMARY KEY,
  customer_id text,
  full_name   text,
  email       text,
  national_id text,
  status      text,
  created_at  timestamp,
  updated_at  timestamp,
  extra_json  text
);

-- ----------------------------------------------------------------------------
-- 3. TRANSACTIONS - Transaction Ledger
-- ----------------------------------------------------------------------------

-- Transaction ledger by account and day - primary transaction log
CREATE TABLE IF NOT EXISTS transactions (
  account_id               text,
  event_date               date,
  event_ts                 timestamp,
  tx_id                    uuid,
  transfer_id              uuid,         -- Links P2P transfer pairs
  sender_id                text,         -- For P2P transfers
  receiver_id              text,         -- For P2P transfers
  direction                text,         -- 'DEBIT' / 'CREDIT'
  counterparty_account_id  text,
  amount                   decimal,
  currency                 text,
  merchant                 text,
  transaction_type         text,         -- p2p_transfer, cash_in, cash_out, etc.
  status                   text,         -- completed, pending, failed
  description              text,
  extra_json               text,
  created_at               timestamp,
  PRIMARY KEY ((account_id, event_date), event_ts, tx_id)
) WITH CLUSTERING ORDER BY (event_ts DESC, tx_id ASC)
  AND compaction = {
    'class':'TimeWindowCompactionStrategy',
    'compaction_window_unit':'DAYS',
    'compaction_window_size':'1'
  };

-- ----------------------------------------------------------------------------
-- 4. P2P TRANSFERS - Peer-to-Peer Transfer Management
-- ----------------------------------------------------------------------------

-- Transfer master record
CREATE TABLE IF NOT EXISTS transfers_by_id (
  transfer_id   uuid PRIMARY KEY,
  from_account  text,
  to_account    text,
  amount        decimal,
  currency      text,
  created_at    timestamp,
  status        text,         -- PENDING/SETTLED/FAILED/REVERSED
  extra_json    text
);

-- Transfer idempotency - prevent duplicate transfers
CREATE TABLE IF NOT EXISTS transfer_dedup (
  from_account       text,
  client_transfer_id text,
  transfer_id        uuid,
  created_at         timestamp,
  PRIMARY KEY ((from_account), client_transfer_id)
);

-- P2P transactions by account pair and day
CREATE TABLE IF NOT EXISTS tx_by_account_pair_day (
  from_account  text,
  to_account    text,
  event_date    date,
  event_ts      timestamp,
  tx_id         uuid,
  transfer_id   uuid,
  amount        decimal,
  currency      text,
  status        text,
  extra_json    text,
  PRIMARY KEY ((from_account, to_account, event_date), event_ts, tx_id)
) WITH CLUSTERING ORDER BY (event_ts DESC, tx_id ASC)
  AND compaction = {
    'class':'TimeWindowCompactionStrategy',
    'compaction_window_unit':'DAYS',
    'compaction_window_size':'1'
  };

-- P2P transactions by account pair and month
CREATE TABLE IF NOT EXISTS p2p_tx_by_account_pair_month (
  party1_account_id text,
  party2_account_id text,
  month_yyyymm      int,        -- 202501, 202502, ...
  event_ts          timestamp,
  tx_id             uuid,
  transfer_id       uuid,
  from_account      text,
  to_account        text,
  amount            decimal,
  currency          text,
  status            text,
  extra_json        text,
  PRIMARY KEY ((party1_account_id, party2_account_id, month_yyyymm), event_ts, tx_id)
) WITH CLUSTERING ORDER BY (event_ts DESC, tx_id ASC);

-- P2P transactions by customer pair and month
CREATE TABLE IF NOT EXISTS p2p_tx_by_customer_pair_month (
  party1_customer_id text,
  party2_customer_id text,
  month_yyyymm       int,
  event_ts           timestamp,
  tx_id              uuid,
  transfer_id        uuid,
  from_customer_id   text,
  to_customer_id     text,
  from_account       text,
  to_account         text,
  amount             decimal,
  currency           text,
  status             text,
  extra_json         text,
  PRIMARY KEY ((party1_customer_id, party2_customer_id, month_yyyymm),event_ts, tx_id)
) WITH CLUSTERING ORDER BY (event_ts DESC, tx_id ASC);

-- ----------------------------------------------------------------------------
-- 5. KPIs - Key Performance Indicators
-- ----------------------------------------------------------------------------

-- Monthly KPIs - system-wide metrics
CREATE TABLE IF NOT EXISTS kpi_monthly (
  month_yyyymm int,
  metric      text,           -- total_volume, num_transactions, avg_amount, etc.
  value       double,
  updated_at  timestamp,
  PRIMARY KEY ((month_yyyymm), metric)
);

-- Monthly KPIs per account
CREATE TABLE IF NOT EXISTS kpi_monthly_by_account (
  account_id   text,
  month_yyyymm int,
  metric       text,   -- total_amount, num_tx, avg_ticket, fail_rate, ...
  value        double,
  updated_at   timestamp,
  PRIMARY KEY ((account_id, month_yyyymm), metric)
);

-- ============================================================================
-- AUDIT KEYSPACE: bank_trading_audit (Logging & Compliance)
-- ============================================================================
USE bank_trading_audit;

-- API call logging - for audit trail and monitoring
CREATE TABLE IF NOT EXISTS api_calls (
  day        date,
  ts         timestamp,
  endpoint   text,
  method     text,
  status     int,
  account_id text,
  client_ip  text,
  extra_json text,
  PRIMARY KEY ((day), ts, endpoint)
) WITH CLUSTERING ORDER BY (ts DESC);

-- NOTES
-- ============================================================================
-- 1. All timestamps are stored in UTC
-- 2. Use SimpleStrategy for dev/test, NetworkTopologyStrategy for production
-- 3. TimeWindowCompactionStrategy is used for time-series data (transactions)
-- 4. All monetary amounts use 'decimal' type for precision
-- 5. extra_json fields allow flexible schema evolution
-- 6. Partition keys are chosen for optimal query patterns
-- 7. Reserved field in balances table supports transaction locking for ACID
-- 8. Multiple customer lookup tables enable fast queries by different identifiers
